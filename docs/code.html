Reusable help components

The problem

New users did not know what to do and could not find help. Help text was being rewritten on every page.

What the code does

Five Blade components (help, steps, tip, next-step, empty-state) plus a small script and stylesheet. A page writes one line, for example <x-ux.help tone="tip">, and gets a consistent callout. A callout can be closed and stays closed (remembered in the browser). Pages that save as you type get one shared "Saving... / All changes saved / 2 not saved" indicator with a retry button, and the browser warns before leaving with unsaved work.

Good parts to notice

  • Props with defaults, so the common case is one line.
  • role="note", aria-label, aria-current and role="status" so assistive technology can follow them.
  • localStorage wrapped in try/catch so private browsing never breaks a page.
  • One save-state object (uxSave) instead of every page inventing its own.

<script>
(function () {
    // "?" tooltips: hover on a laptop, tap on a phone
    function initTips(root) {
        if (!window.bootstrap) { return; }
        (root || document).querySelectorAll('[data-ux-tip]:not([data-ux-ready])').forEach(function (el) {
            el.setAttribute('data-ux-ready', '1');
            new bootstrap.Popover(el, { trigger: 'hover focus click', placement: 'top', html: false, content: el.getAttribute('data-ux-tip'), container: 'body' });
        });
    }
    document.addEventListener('DOMContentLoaded', function () { initTips(); });

    // Callouts that can be closed and stay closed
    function seen(key) { try { return localStorage.getItem('ux-hide-' + key) === '1'; } catch (e) { return false; } }
    document.addEventListener('DOMContentLoaded', function () {
        document.querySelectorAll('.ux-help[data-ux-key]').forEach(function (box) {
            if (seen(box.getAttribute('data-ux-key'))) { box.remove(); }
        });
    });
    document.addEventListener('click', function (e) {
        var x = e.target.closest('.ux-x');
        if (!x) { return; }
        var box = x.closest('.ux-help');
        try { localStorage.setItem('ux-hide-' + box.getAttribute('data-ux-key'), '1'); } catch (err) {}
        box.remove();
    });

    // "Saving... / All changes saved / 2 not saved" for pages that save as you type.
    // uxSave.begin() when a save starts; uxSave.ok(key) when it worked; uxSave.fail(key, message, retryFunction) when it did not.
    var saveEl = null, saveFade = null, savePending = 0, saveFailed = new Map();
    function renderSave() {
        if (!saveEl) { saveEl = document.createElement('div'); saveEl.id = 'uxSaveState'; saveEl.setAttribute('role', 'status'); document.body.appendChild(saveEl); }
        clearTimeout(saveFade);
        if (savePending > 0) {
            saveEl.className = 'ux-savestate ux-busy'; saveEl.innerHTML = '<i class="fas fa-circle-notch fa-spin"></i> Saving...';
        } else if (saveFailed.size > 0) {
            saveEl.className = 'ux-savestate ux-bad';
            saveEl.innerHTML = '<i class="fas fa-triangle-exclamation"></i> ' + saveFailed.size + ' not saved <button type="button" data-ux-retry>Try again</button>';
        } else {
            saveEl.className = 'ux-savestate ux-ok'; saveEl.innerHTML = '<i class="fas fa-check"></i> All changes saved';
            saveFade = setTimeout(function () { saveEl.classList.add('ux-fade'); }, 2500);
        }
    }
    window.uxSave = {
        begin: function () { savePending++; renderSave(); },
        ok: function (key) { savePending = Math.max(0, savePending - 1); if (key !== undefined) { saveFailed.delete(key); } renderSave(); },
        fail: function (key, message, retry) {
            savePending = Math.max(0, savePending - 1);
            saveFailed.set(key, retry || null);
            if (message) { uxToast(message, 'bad'); }
            renderSave();
        },
        retryAll: function () { var fns = Array.from(saveFailed.values()); saveFailed.clear(); renderSave(); fns.forEach(function (f) { if (f) { f(); } }); }
    };
    document.addEventListener('click', function (e) { if (e.target.closest('[data-ux-retry]')) { window.uxSave.retryAll(); } });
    window.addEventListener('beforeunload', function (e) {
        if (savePending > 0 || saveFailed.size > 0) { e.preventDefault(); e.returnValue = ''; }
    });

    // Friendly messages instead of browser pop-ups: uxToast('Saved', 'ok') or uxToast('Could not save', 'bad')
    window.uxToast = function (message, tone) {
        var wrap = document.getElementById('uxToasts');
        if (!wrap) { wrap = document.createElement('div'); wrap.id = 'uxToasts'; document.body.appendChild(wrap); }
        var t = document.createElement('div');
        t.className = 'ux-toast' + (tone ? ' ux-' + tone : '');
        t.setAttribute('role', 'status');
        t.textContent = message;
        wrap.appendChild(t);
        setTimeout(function () { t.remove(); }, tone === 'bad' ? 7000 : 3500);
    };
})();
</script>

Get Started card and the setup checklist

The problem

Schools said "after registering I don't know what to do next". A seven-step wizard put Students last.

What the code does

One class, SetupProgress, is the single source of truth for the checklist, the next step, the percentage and the minutes left. The Get Started card, the wizard banner, the sidebar and the "where next" links all read it, so they can never disagree.

Good parts to notice

  • Order chosen for the user: classes, then students (what they came to do), then rules.
  • Each step carries its own plain-language explanation and time estimate.
  • The card hides itself when setup is finished or the admin chooses "I will set up later".
  • White-labelled schools get no StudyPat support link.

<?php

namespace App\Services;

use App\Models\AcademicSession;
use App\Models\GradingSystem;
use App\Models\School;
use App\Models\SchoolClass;
use App\Models\ScoreDivision;
use App\Models\Staff;
use App\Models\Student;
use App\Models\Subject;

/**
 * The first-time setup of a school, in the order that is easiest for a new admin: classes first, then the students
 * (what they came to do), then subjects and teachers, then the two "rules" (how grades are given and how a subject's
 * marks are split). One list used by the dashboard's Get Started card, the setup wizard banner, the wizard's own
 * "where to next" and the sidebar, so they can never disagree about what is done and what comes next.
 */
class SetupProgress
{
    /** @return array{steps: array<string, array>, done: int, total: int, percent: int, next: ?array, incomplete: bool, minutesLeft: int} */
    public function forSchool(School $school): array
    {
        $id = $school->id;
        $url = fn (string $route) => route($route, ['subdomain' => $school->subdomain, 'wizard' => 1]);

        $steps = [
            'classes' => [
                'label' => 'Classes & Arms', 'title' => 'Add your classes', 'route' => 'tenant.academics.classes', 'minutes' => 2,
                'explain' => 'Add the classes your school has, such as Primary 1 or JSS 2. If a class has sections (A, B, C), add them as arms.',
                'done' => SchoolClass::where('school_id', $id)->exists(),
            ],
            'students' => [
                'label' => 'Students', 'title' => 'Add your students', 'route' => 'tenant.people.students', 'minutes' => 5,
                'explain' => 'Add your students one by one, or upload a whole class from a spreadsheet. Each parent gets a login automatically.',
                'done' => Student::where('school_id', $id)->exists(),
            ],
            'subjects' => [
                'label' => 'Subjects', 'title' => 'Add your subjects', 'route' => 'tenant.academics.subjects.create', 'minutes' => 3,
                'explain' => 'Add the subjects each class studies. You can add many at once, and assign them to teachers later.',
                'done' => Subject::where('school_id', $id)->exists(),
            ],
            'staff' => [
                'label' => 'Teachers', 'title' => 'Add your teachers', 'route' => 'tenant.people.staff.create', 'minutes' => 3,
                'explain' => 'Add your teachers and give each a login. Tick the subjects they teach so they see them when they sign in.',
                'done' => Staff::where('school_id', $id)->exists(),
            ],
            'grading' => [
                'label' => 'Grading', 'title' => 'Choose how grades are given', 'route' => 'tenant.academics.grading', 'minutes' => 1,
                'explain' => 'Pick a ready-made scale (such as A1 to F9) with one click, or set your own. Report cards use it to show a grade.',
                'done' => GradingSystem::where('school_id', $id)->exists(),
            ],
            'divisions' => [
                'label' => 'Score split', 'title' => 'Choose how scores are split', 'route' => 'tenant.results.score-divisions', 'minutes' => 1,
                'explain' => 'Pick how each subject is scored, for example CA 30 and Exam 70 (together they make 100), or set your own.',
                'done' => ScoreDivision::where('school_id', $id)->exists(),
            ],
            'sessions' => [
                'label' => 'Session & Terms', 'title' => 'Set your session and terms', 'route' => 'tenant.academics.sessions_terms', 'minutes' => 1,
                'explain' => 'Create the school year (for example 2026/2027) with its terms, and mark the current one. Scores and fees are recorded against it.',
                'done' => AcademicSession::where('school_id', $id)->exists(),
            ],
        ];

        $next = null;
        $done = 0;
        $minutesLeft = 0;
        foreach ($steps as $key => &$step) {
            $step['key'] = $key;
            $step['url'] = $url($step['route']);
            if ($step['done']) {
                $done++;
            } else {
                $minutesLeft += $step['minutes'];
                $next = $next ?? $step;
            }
        }
        unset($step);

        $total = count($steps);

        return [
            'steps' => $steps,
            'done' => $done,
            'total' => $total,
            'percent' => (int) round($done / $total * 100),
            'next' => $next,
            'incomplete' => $next !== null,
            'minutesLeft' => $minutesLeft,
        ];
    }

    public function incomplete(School $school): bool
    {
        return $this->forSchool($school)['incomplete'];
    }
}

// StudyPat installable app - service worker
// Pages always come from the network (they are personal and carry security tokens); only the "you are offline" page
// and static files (fonts, icons, logos, libraries) are kept on the phone, so the app opens fast and fails politely.
const VERSION = '__VERSION__';
const SHELL = 'sp-shell-' + VERSION;
const STATIC = 'sp-static-' + VERSION;
const OFFLINE_URL = '/offline';
const CDN_HOSTS = ['cdn.jsdelivr.net', 'cdnjs.cloudflare.com', 'fonts.googleapis.com', 'fonts.gstatic.com', 'code.jquery.com'];

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(SHELL)
      .then((cache) => cache.add(new Request(OFFLINE_URL, { cache: 'reload' })))
      .then(() => self.skipWaiting())
  );
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys()
      .then((keys) => Promise.all(keys.filter((k) => k !== SHELL && k !== STATIC).map((k) => caches.delete(k))))
      .then(() => self.clients.claim())
  );
});

async function cacheFirst(request) {
  const cache = await caches.open(STATIC);
  const hit = await cache.match(request);
  if (hit) return hit;
  const response = await fetch(request);
  if (response && (response.ok || response.type === 'opaque')) cache.put(request, response.clone());
  return response;
}

async function staleWhileRevalidate(request) {
  const cache = await caches.open(STATIC);
  const hit = await cache.match(request);
  const refresh = fetch(request).then((response) => {
    if (response && response.ok) cache.put(request, response.clone());
    return response;
  }).catch(() => hit);
  return hit || refresh;
}

self.addEventListener('fetch', (event) => {
  const request = event.request;
  if (request.method !== 'GET' || request.headers.has('range')) return;
  const url = new URL(request.url);

  if (request.mode === 'navigate') {
    event.respondWith(fetch(request).catch(() => caches.match(OFFLINE_URL)));
    return;
  }
  if (CDN_HOSTS.includes(url.hostname)) {
    event.respondWith(cacheFirst(request).catch(() => fetch(request)));
    return;
  }
  if (url.origin === self.location.origin &&
      (url.pathname.startsWith('/storage/') || url.pathname.startsWith('/pwa/') ||
       /\.(png|jpe?g|svg|webp|ico|woff2?)$/i.test(url.pathname))) {
    event.respondWith(staleWhileRevalidate(request));
  }
});

The exam screen: save queue, timer, safety

The problem

In an exam hall the network is poor and a lost answer is serious. The candidate must always know whether their work is saved.

What the code does

Every answer goes into a queue and is re-sent until the server says "saved". Each answer carries a version, so a newer edit made while an older one is in flight is never lost. Failures retry with exponential back-off (2 s, 4 s … up to 15 s). A status line shows "All answers saved", "Saving…" or "2 not saved, retrying". The timer counts to a deadline and is re-synchronised with the server clock every 20 seconds. A phone is kept awake, and leaving the tab is reported to the invigilator.

Good parts to notice

  • One shape for every question, whatever the server's letter case.
  • Anything that is not JSON (a login page, an error page) counts as a failed save, so the answer is kept and retried.
  • Answers the server will never accept are dropped instead of retried forever.
  • A lost violation report can never break the exam.
  • The Screen Wake Lock API with a safe fallback.

/// <summary>
/// Turns the last few delay measurements to the company computer into a plain-words grade, so a user (and support) can tell at a glance
/// whether "slow" or "hanging" comes from the connection and not from the program. A null sample means the answer never came back.
/// What delay feels like: up to 50 ms is hard to tell from sitting at the computer, up to 100 ms is comfortable, up to 200 ms is usable
/// but the typing and clicking lag is noticeable, above that it feels slow. Lost answers (packet loss) cause the freezes.
/// </summary>
public static class ConnectionQuality
{
    public static string Grade(IReadOnlyList<int?> samples, out int avgMs, out int lossPercent)
    {
        avgMs = 0; lossPercent = 0;
        if (samples.Count == 0) return "unknown";
        var ok = samples.Where(x => x.HasValue).Select(x => x!.Value).ToList();
        lossPercent = (samples.Count - ok.Count) * 100 / samples.Count;
        if (ok.Count == 0) return "silent";
        avgMs = (int)Math.Round(ok.Average());
        var jitter = ok.Max() - ok.Min();
        var g = avgMs <= 50 ? 0 : avgMs <= 100 ? 1 : avgMs <= 200 ? 2 : 3;     // 0 excellent .. 3 poor
        if (jitter > 150) g = Math.Max(g, 2);                                    // delay that jumps around feels bad even when the average is fine
        // only 8 measurements are kept, so one lost message is already 12%: a single hiccup is "Fair" at worst; "Poor" needs several lost
        if (lossPercent > 25) g = 3; else if (lossPercent > 0) g = Math.Max(g, 2);
        return new[] { "Excellent", "Good", "Fair", "Poor" }[g];
    }

    public static string Describe(IReadOnlyList<int?> samples)
    {
        var grade = Grade(samples, out var avg, out var loss);
        switch (grade)
        {
            case "unknown": return "";
            case "silent": return "Connection quality: cannot measure (the company computer does not answer a test message).";
        }
        var text = $"Connection quality: {grade}, {avg} ms delay" + (loss > 0 ? $", {loss}% of test messages lost" : "") + ".";
        if (grade == "Fair") text += " It may feel a little slow. A cable instead of Wi-Fi, or moving closer to the router, usually helps.";
        if (grade == "Poor") text += " It will feel slow or freeze. Try a cable instead of Wi-Fi; if it stays poor tell your administrator.";
        return text;
    }
}
